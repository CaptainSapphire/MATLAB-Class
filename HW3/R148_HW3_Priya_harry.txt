%% Problem 3 — Rounding for an engineering decision (6 points)
% A steel rod is sold only in whole 0.1-m lengths. A design requires 3.24 m.
% Calculate the minimum length that must be purchased. Use the appropriate
% rounding function and explain your choice in a comment.
%
% Your code:
required_length_m = 3.24;
increment_m = 0.1;

% Scale to tenths, round up with ceil to ensure sufficient material, then scale back
purchased_length_m = ceil(required_length_m / increment_m) * increment_m;

fprintf('Problem 3: Minimum purchased rod length = %.2f m\n\n', purchased_length_m);

% Comment:
% ceil (ceiling) is chosen because purchasing any less than 3.24 m would leave
% the rod too short for the design; rounding up guarantees meeting or exceeding specifications.

%% Problem 8 — Simulated measurements (8 points)
% Generate 20 simulated temperature readings with a mean of 22 C and a
% standard deviation of 0.4 C. Use randn.
% (a) Calculate the mean and standard deviation of your simulated readings.
% (b) Create a histogram with a title, labeled axes, and grid lines.
% (c) In a comment, explain why your simulated mean may not be exactly 22 C.

% Your code:
target_mean_C = 22;
target_std_C = 0.4;
num_samples = 20;

% Simulated readings formula: mu + sigma * randn(1, N)
simulated_temp_C = target_mean_C + target_std_C * randn(1, num_samples);

% (a) Sample mean and standard deviation
sample_mean_C = mean(simulated_temp_C);
sample_std_C = std(simulated_temp_C);

fprintf('Problem 8:\n');
fprintf('  (a) Sample Mean: %.3f C | Sample Std Dev: %.3f C\n\n', sample_mean_C, sample_std_C);

% (b) Histogram
figure(1);
histogram(simulated_temp_C, 6);
title('Simulated Temperature Distribution (N = 20)');
xlabel('Temperature (°C)');
ylabel('Frequency (Count)');
grid on;

% (c) Explanation:
% A finite random sample (N = 20) exhibits natural statistical variation (sampling error).
% By the Law of Large Numbers, the sample mean converges toward the population mean (22 °C)
% only as the sample size N approaches infinity.
